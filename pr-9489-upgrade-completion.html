ReconcileService Implementation (Lines 33-81)

func ReconcileService(svc *corev1.Service, strategy *hyperv1.ServicePublishingStrategy, owner *metav1.OwnerReference, apiServerServicePort int, apiAllowedCIDRBlocks []string, hcp *hyperv1.HostedControlPlane) error { // CREATE vs UPDATE detection: ResourceVersion empty only on CREATE isCreate := svc.ResourceVersion == "" // ... configuration setup ... switch strategy.Type { case hyperv1.LoadBalancer: // AWS requires the load balancer type annotation to remain unchanged // after Service creation. The cluster-wide ValidatingAdmissionPolicy // "openshift-cloud-controller-manager-cloud-provider-aws" (OCPBUGS-16728) // blocks any UPDATE that adds, removes, or changes this annotation. // Seed it only on CREATE, which the policy does not match. if isCreate && hcp.Spec.Platform.Type == hyperv1.AWSPlatform { svc.Annotations[AWSNLBAnnotation] = "nlb" } // ... rest of configuration ... } return nil }

Why This Works

  • ResourceVersion == "": Only true on first creation, false on all subsequent updates
  • Conditional seeding: Annotation added only when isCreate is true
  • Avoids VAP block: CREATE operations don't match the AWS policy filter; UPDATE operations do
  • Prevents modification: Future reconciliation cycles skip annotation modification since isCreate is false
